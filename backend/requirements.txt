# BioVision Python Backend Requirements
# Python 3.10+ required (3.11 recommended).
#
# Recommended install path:
#   Windows:      setup.bat
#   macOS/Linux:  bash setup.sh
#
# These wrappers call setup_backend.py, which:
# - creates venv
# - selects the correct torch build (CUDA/MPS/CPU)
# - installs this requirements file
# - installs the pinned CLIP text encoder required by YOLO-World prompts
#
# Manual install (after activating venv):
#   pip install -r backend/requirements.txt

# ------------------------------------------------------------------------------
# Core runtime dependencies
# ------------------------------------------------------------------------------
# Keep PyTorch's NumPy bridge working on every release target, including Intel
# macOS. OpenCV 4.12+ requires NumPy 2 on Python 3.9+, while the supported
# PyTorch builds still require the NumPy 1.x ABI for tensor conversion.
opencv-python==4.11.0.86
numpy==1.26.4
Pillow>=10.0.0

# Landmark model training / inference
# NOTE: dlib needs CMake and a C++ build toolchain.
#   Windows: Visual Studio Build Tools + CMake
#   macOS:   brew install cmake
#   Linux:   apt install cmake build-essential
dlib>=19.24.0

# Detection + SAM2/YOLO pipeline
ultralytics==8.4.14

# YOLO-World resolves arbitrary user prompts through CLIP. Pin the exact
# Ultralytics-compatible source revision so release builds cannot omit it or
# silently change when upstream moves.
clip @ git+https://github.com/ultralytics/CLIP.git@68dce32140994dfcb645a1320c4ebdc034fc19fd

# Orientation PCA quality path (fallback exists if missing, but this improves it)
scikit-learn>=1.3.0

# System capability checks (RAM, gating)
psutil>=5.9.0

# ------------------------------------------------------------------------------
# Managed separately by setup_backend.py
# ------------------------------------------------------------------------------
# torch / torchvision / torchaudio:
#   Do NOT add here; setup_backend.py installs the correct CUDA/MPS/CPU build.
#
# CLIP is pinned above because it is a required YOLO-World runtime dependency.
